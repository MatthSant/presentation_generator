CSS a sobrescrever

/* 1. Variáveis de raiz */
:root {
  --bg:     #FFFFFF;
  --border: #E5E7EB;
  --gray:   #6B7280;
  --gray2:  #9CA3AF;
}

/* 2. Slide title (hardcoded #fff → dark) */
h1.slide-title { color: #111827; }

/* 3. Superfícies semi-transparentes */
.card { background: rgba(0,0,0,.025); border: 1px solid rgba(0,0,0,.09); }
.card-title { color: #111827; }

.mr  { border: 1px solid rgba(0,0,0,.10); }
.mi  { border-right: 1px solid rgba(0,0,0,.08); }
.mv  { color: #111827; }

/* 4. Highlights (texto hardcoded #fff → dark) */
.hl, .hl-g, .hl-a, .hl-r { color: #374151; }

/* 5. Bullets */
.def-bullets li { color: #374151; }
ul.bul li       { color: #374151; }
ul.bul li strong { color: #111827; }

/* 6. Find blocks */
.find-block  { border-color: rgba(0,0,0,.08); }
.find-title  { color: #111827; }
.find-note   { color: #374151; }

/* 7. .c-w deixa de ser usado — use .c-p, .c-g, etc. */

O id deve começar com chart- e ser único no documento. A altura é obrigatória via JS (height:) — CSS flex:1 não funciona no ApexCharts.
<div class="chart-wrap" id="chart-nome"></div>

<!-- No bloco chartDefs: -->
'chart-nome': {
  type: 'bar',
  height: 240,
  series: [
    { name: 'Série A', data: [42, 58, 71, 89] },
    { name: 'Série B', data: [28, 35, 49, 61] },
  ],
  categories: ['2022', '2023', '2024', '2025'],
  colors: ['#8B5CF6', '#10B981'],
},

Parâmetros úteis

dataLabels: {
  enabled: true,
  formatter: v => v + '%',          // formatar valor exibido na barra
  offsetY: -4,                       // ajuste vertical do rótulo
  style: { fontSize: '11px', colors: ['#fff'] },
},
plotOptions: {
  bar: {
    borderRadius: 4,                 // arredondar pontas
    columnWidth: '55%',              // largura das colunas (padrão 70%)
    distributed: false,              // true → cor diferente por barra
    dataLabels: { position: 'top' }, // 'top' | 'center' | 'bottom'
  },
},
stroke: { show: true, width: 2, colors: ['transparent'] }, // borda nas barras
xaxis: { title: { text: 'Período', style: { color: '#9CA3AF' } } },
yaxis: { min: 0, max: 100, tickAmount: 5 },
annotations: {                       // linha de referência horizontal

Correlação entre duas variáveis contínuas. Use buildTrendSeries(pts, type) para sobrepor linha de tendência com R² automático. Tipos suportados: 'linear', 'exp', 'log', 'pow'.

Com linha de tendência (linear)
/* ── Helper: regressão + linha de tendência ── */
function calcR2(pts, fn) {
  const my = pts.reduce((s,[,y])=>s+y,0)/pts.length;
  return 1 - pts.reduce((s,[x,y])=>s+(y-fn(x))**2,0)
             / pts.reduce((s,[,y])=>s+(y-my)**2,0);
}
function trendLinear(pts) {
  const n=pts.length, sx=pts.reduce((s,[x])=>s+x,0), sy=pts.reduce((s,[,y])=>s+y,0),
        sxy=pts.reduce((s,[x,y])=>s+x*y,0), sx2=pts.reduce((s,[x])=>s+x*x,0);
  const m=(n*sxy-sx*sy)/(n*sx2-sx*sx), b=(sy-m*sx)/n;
  return { fn:x=>m*x+b, label:'Linear' };
}
function trendExp(pts) {  /* y = a·eˣᵇ  — requer y > 0 */
  const lp=pts.filter(([,y])=>y>0).map(([x,y])=>[x,Math.log(y)]);
  const {fn:lf}=trendLinear(lp); return { fn:x=>Math.exp(lf(x)), label:'Exponencial' };
}
function trendLog(pts) {  /* y = a·ln(x) + b  — requer x > 0 */
  const lp=pts.filter(([x])=>x>0).map(([x,y])=>[Math.log(x),y]);
  const {fn:lf}=trendLinear(lp); return { fn:x=>lf(Math.log(x)), label:'Logarítmica' };
}
function trendPow(pts) {  /* y = a·xᵇ  — requer x,y > 0 */
  const lp=pts.filter(([x,y])=>x>0&&y>0).map(([x,y])=>[Math.log(x),Math.log(y)]);
  const {fn:lf}=trendLinear(lp); return { fn:x=>Math.exp(lf(Math.log(x))), label:'Potência' };
}
